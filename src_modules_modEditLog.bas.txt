Public Function IsAdminUser(ByVal ws As Worksheet) As Boolean
    Dim raw As String
    Dim adminList() As String
    Dim i As Long
    Dim u1 As String, u2 As String
    Dim admin As String

    raw = CStr(ws.Range(EDITLOG_ADMIN_CELL).value)
    raw = Trim$(raw)
    If raw = "" Then Exit Function

    ' ���K���F�������{�S�p�X�y�[�X�����p�A�Z�~�R�������J���}
    raw = LCase$(raw)
    raw = Replace(raw, "�@", " ")
    raw = Replace(raw, ";", ",")
    ' �󔒂���؂�Ƃ��Ĉ���
    raw = Replace(raw, " ", ",")

    adminList = Split(raw, ",")

    u1 = LCase$(Trim$(Environ$("USERNAME")))
    u2 = LCase$(Trim$(Application.UserName))

    For i = LBound(adminList) To UBound(adminList)
        admin = Trim$(adminList(i))
        If admin <> "" Then
            If admin = u1 Or admin = u2 Then
                IsAdminUser = True
                Exit Function
            End If
        End If
    Next i
End Function

Public Function GetHeaderColumn( _
        ByVal ws As Worksheet, _
        ByVal headerText As String) As Long

    Dim lastCol As Long
    Dim col As Long

    On Error GoTo ExitFunc

    lastCol = ws.Cells(EDITLOG_HEADER_ROW, ws.Columns.Count).End(xlToLeft).Column

    For col = 1 To lastCol
        If CStr(ws.Cells(EDITLOG_HEADER_ROW, col).value) = headerText Then
            GetHeaderColumn = col
            Exit Function
        End If
    Next col

ExitFunc:
End Function

'  ���C�������F�ҏW���O�����F�h��
'  �E�Ǘ��҂��u�ŏI�ҏW�ҁ^�����v�͍X�V

Public Sub LogEditAndHighlight(ByVal Target As Range)
    Dim ws As Worksheet
    Dim rngData As Range
    Dim rngChange As Range
    Dim colEditor As Long
    Dim colEditTime As Long
    Dim r As Range, c As Range
    Dim userDisp As String
    Dim isAdmin As Boolean
    Dim eventsBefore As Boolean
    Dim screenBefore As Boolean
    Dim loginName As String

    On Error GoTo ErrHandler

    If Target Is Nothing Then Exit Sub

    Set ws = Target.Worksheet
    If ws.name <> EDITLOG_SHEET_NAME Then Exit Sub

    ' �f�[�^�s�i4�s�ڈȍ~�j�̂ݑΏ�
    Set rngData = ws.Rows(EDITLOG_DATA_START & ":" & ws.Rows.Count)
    Set rngChange = Intersect(Target, rngData)
    If rngChange Is Nothing Then Exit Sub

    ' ���o����擾
    colEditor = GetHeaderColumn(ws, EDITLOG_HDR_EDITOR)
    colEditTime = GetHeaderColumn(ws, EDITLOG_HDR_EDITTIME)
    If colEditor = 0 Or colEditTime = 0 Then Exit Sub

    ' �Ǘ��Ҕ���
    isAdmin = IsAdminUser(ws)

    ' ���O�C�����i���̃��[�U�[���j
    loginName = Application.UserName

    ' �\���p���[�U�[���i���O�Ɏc�����j
    ' �E�Ǘ��҂Ȃ� K2�Z���̖��O���g��
    ' �E����ȊO�͌��̃��[�U�[�������̂܂܎g��
    If isAdmin Then
        userDisp = CStr(ws.Range(EDITLOG_ADMIN_CELL).value)
        If Trim$(userDisp) = "" Then
            userDisp = loginName   ' �O�̂��ߋ�Ȃ烍�O�C�����Ƀt�H�[���o�b�N
        End If
    Else
        userDisp = loginName
    End If

    ' ���̏�Ԃ�ۑ�
    eventsBefore = Application.EnableEvents
    screenBefore = Application.ScreenUpdating

    Application.EnableEvents = False
    Application.ScreenUpdating = False

    ' �ŏI�ҏW�ҁE�������X�V�i�������̓Z�����̐ݒ�𑸏d�j
    For Each r In rngChange.Rows
        ws.Cells(r.row, colEditor).value = userDisp
        ws.Cells(r.row, colEditTime).value = Now
    Next r

    ' ��ʃ��[�U�[�������F�h��
    If Not isAdmin Then
        For Each c In rngChange.Cells
            If c.Column <> colEditor And c.Column <> colEditTime Then
                c.Interior.Color = vbYellow
            End If
        Next c
    End If

ExitHandler:
    ' �K�����̏�Ԃɖ߂�
    On Error Resume Next
    Application.EnableEvents = eventsBefore
    Application.ScreenUpdating = screenBefore
    Exit Sub

ErrHandler:
    ' �K�v�Ȃ炱���� Debug.Print �⃍�O�o�͂���
    ' Debug.Print "LogEditAndHighlight Error: " & Err.Number & " - " & Err.Description
    Resume ExitHandler
End Sub

'  �_�肫�ύX�҂̃V�[�g����ID�ɐ������镶��������擾

Public Function FindUserNameFromID(ByVal userID As String) As String
    Dim wsLedger As Worksheet
    Dim lastRow As Long
    Dim r As Long
    Dim idNormalized As String
    Dim currentID As String

    On Error GoTo ExitFunc

    idNormalized = UCase$(Trim$(userID))
    If idNormalized = "" Then GoTo ExitFunc

    Set wsLedger = ThisWorkbook.Worksheets("�_�肫�ύX��")
    lastRow = wsLedger.Cells(wsLedger.Rows.Count, "A").End(xlUp).Row
    If lastRow < 2 Then GoTo ExitFunc

    For r = 2 To lastRow
        currentID = UCase$(Trim$(CStr(wsLedger.Cells(r, "A").value)))
        If currentID = idNormalized Then
            FindUserNameFromID = CStr(wsLedger.Cells(r, "B").value)
            Exit For
        End If
    Next r

ExitFunc:
End Function

Public Sub RegisterIDIfNew(ByVal userID As String)
    Dim wsLedger As Worksheet
    Dim lastRow As Long
    Dim r As Long
    Dim idNormalized As String
    Dim currentID As String

    On Error GoTo ExitProc

    idNormalized = UCase$(Trim$(userID))
    If idNormalized = "" Then GoTo ExitProc

    Set wsLedger = ThisWorkbook.Worksheets("�_�肫�ύX��")
    lastRow = wsLedger.Cells(wsLedger.Rows.Count, "A").End(xlUp).Row
    If lastRow < 2 Then lastRow = 1

    For r = 2 To lastRow
        currentID = UCase$(Trim$(CStr(wsLedger.Cells(r, "A").value)))
        If currentID = idNormalized Then GoTo ExitProc
    Next r

    wsLedger.Cells(lastRow + 1, "A").value = userID

ExitProc:
End Sub

Public Function ResolveUserDisplayName(ByVal userID As String) As String
    Dim nameResolved As String

    nameResolved = FindUserNameFromID(userID)
    If Trim$(nameResolved) <> "" Then
        ResolveUserDisplayName = nameResolved
    Else
        ResolveUserDisplayName = userID
    End If
End Function
