Option Explicit

'=================================================
' 口座名義 正規化（振込用 半角カタカナ）
'=================================================

'---------------------------------------------
' 法人略語ルール格納用
'---------------------------------------------
Private Type AbbrRule
    term As String          ' 置換対象（元の語）
    headAbbr As String      ' 先頭にあるときの略語
    midAbbr As String       ' 中間にあるときの略語
    tailAbbr As String      ' 末尾にあるときの略語
End Type

'=================================================
' 口座名義 正規化メイン
'=================================================
Public Function NormalizeAccountName(ByVal inputText As String) As String
    Dim result As String
    Dim originalHasSpace As Boolean

    ' 0) 空文字なら即終了
    If Len(inputText) = 0 Then
        NormalizeAccountName = ""
        Exit Function
    End If

    ' 元の文字列にスペースが含まれているか（個人名判定用）
    originalHasSpace = (InStr(inputText, " ") > 0 Or InStr(inputText, "　") > 0)

    ' 改行除去（後続の位置判定を安定させる）
    result = Replace(Replace(inputText, vbCr, ""), vbLf, "")

    ' 1) 全角英数 → 半角英数
    result = StrConv(result, vbNarrow)

    ' 2) ひらがな → カタカナ（全角）
    result = StrConv(result, vbKatakana)

    ' 3) 全角カタカナ → 半角カタカナ
    result = StrConv(result, vbNarrow)

    ' 4) かっこ類を半角へ統一
    result = NormalizeParentheses(result)

    ' 5) 法人格・営業所・事業体の略称統一（位置判定つき）
    result = NormalizeCorpAbbr(result)
    result = NormalizeBranchAbbr(result)
    result = NormalizeBusinessAbbr(result)

    ' 6) 小さいカナ → 大きいカナ
    result = NormalizeSmallKana(result)

    ' 7) 記号の統一（長音・中点）
    result = Replace(result, "ｰ", "-")
    result = Replace(result, "－", "-")
    result = Replace(result, "ー", "-")
    result = Replace(result, "・", ".")
    result = Replace(result, "･", ".")

    ' 8) スペース正規化
    result = NormalizeSpaces(result, originalHasSpace)

    ' 9) 余計なダブルクォート削除＋Trim
    result = Replace(result, """", "")
    result = Application.Trim(result)

    NormalizeAccountName = result
End Function

' テストイメージ（参考・Assertはしない）
' ・株式会社太陽建設工業 → ｶ)ﾀｲﾖｳｹﾝｾﾂｺｳｷﾞﾖｳ
' ・太陽建設工業株式会社 → ﾀｲﾖｳｹﾝｾﾂｺｳｷﾞﾖｳ(ｶ
' ・医療法人社団〇〇会   → ｲﾘ)ｼﾔﾀﾞﾝ〇〇ｶｲ（位置で略語が変わる）

'---------------------------------------------
' かっこ類を半角へ統一
'---------------------------------------------
Private Function NormalizeParentheses(ByVal s As String) As String
    Dim r As String
    r = s
    r = Replace(r, "（", "(")
    r = Replace(r, "）", ")")
    r = Replace(r, "〔", "[")
    r = Replace(r, "〕", "]")
    NormalizeParentheses = r
End Function

'---------------------------------------------
' 位置判定付き置換
'---------------------------------------------
Private Function ReplaceWithPosition(ByVal s As String, ByVal term As String, _
    ByVal headAbbr As String, ByVal midAbbr As String, ByVal tailAbbr As String) As String

    Dim pos As Long
    pos = InStr(1, s, term, vbTextCompare)

    Do While pos > 0
        Dim rep As String
        If pos = 1 Then
            rep = headAbbr                     ' 先頭
        ElseIf pos + Len(term) - 1 = Len(s) Then
            rep = tailAbbr                     ' 末尾
        Else
            rep = midAbbr                      ' 中間
        End If

        s = Left$(s, pos - 1) & rep & Mid$(s, pos + Len(term))
        pos = InStr(pos + Len(rep), s, term, vbTextCompare)
    Loop

    ReplaceWithPosition = s
End Function

'---------------------------------------------
' 法人格略語統一（先頭/中間/末尾で略語形を変える）
'---------------------------------------------
Private Function NormalizeCorpAbbr(ByVal s As String) As String
    Dim rules() As AbbrRule
    rules = BuildCorpRules()

    Dim i As Long, patterns As Variant, p As Variant
    For i = LBound(rules) To UBound(rules)
        patterns = Array(rules(i).term, "(" & rules(i).term & ")")
        If rules(i).term = "株式会社" Then patterns = Array(rules(i).term, "(" & rules(i).term & ")", "(株)", "㈱")
        If rules(i).term = "有限会社" Then patterns = Array(rules(i).term, "(" & rules(i).term & ")", "(有)", "㈲")

        For Each p In patterns
            s = ReplaceWithPosition(s, CStr(p), rules(i).headAbbr, rules(i).midAbbr, rules(i).tailAbbr)
        Next p
    Next i

    NormalizeCorpAbbr = s
End Function

Private Function MakeRule(ByVal term As String, ByVal headAbbr As String, ByVal midAbbr As String, ByVal tailAbbr As String) As AbbrRule
    Dim r As AbbrRule
    r.term = term
    r.headAbbr = headAbbr
    r.midAbbr = midAbbr
    r.tailAbbr = tailAbbr
    MakeRule = r
End Function

'---------------------------------------------
' 小さいカナ → 大きいカナ（半角）
'---------------------------------------------
Private Function NormalizeSmallKana(ByVal s As String) As String
    Dim r As String
    r = s
    r = Replace(r, "ｧ", "ｱ")
    r = Replace(r, "ｨ", "ｲ")
    r = Replace(r, "ｩ", "ｳ")
    r = Replace(r, "ｪ", "ｴ")
    r = Replace(r, "ｫ", "ｵ")
    r = Replace(r, "ｬ", "ﾔ")
    r = Replace(r, "ｭ", "ﾕ")
    r = Replace(r, "ｮ", "ﾖ")
    r = Replace(r, "ｯ", "ﾂ")
    NormalizeSmallKana = r
End Function

'---------------------------------------------
' スペース正規化
' ・元文字列にスペースあり  → 全角→半角＋連続スペースを 1 個に圧縮
' ・元文字列にスペースなし → すべて削除
'---------------------------------------------
Private Function NormalizeSpaces(ByVal s As String, ByVal originalHasSpace As Boolean) As String
    Dim r As String
    r = Replace(s, "　", " ")

    If originalHasSpace Then
        Do While InStr(r, "  ") > 0
            r = Replace(r, "  ", " ")
        Loop
    Else
        r = Replace(r, " ", "")
    End If

    NormalizeSpaces = Trim(r)
End Function

'---------------------------------------------
' 営業所略語（先頭/中間/末尾別）
'---------------------------------------------
Private Function NormalizeBranchAbbr(ByVal s As String) As String
    Dim rules(1 To 2) As AbbrRule
    rules(1) = MakeRule("営業所", "ｴｲ)", "(ｴｲ)", "(ｴｲ")
    rules(2) = MakeRule("出張所", "ｼﾕﾂ)", "(ｼﾕﾂ)", "(ｼﾕﾂ")

    Dim i As Long
    For i = LBound(rules) To UBound(rules)
        s = ReplaceWithPosition(s, rules(i).term, rules(i).headAbbr, rules(i).midAbbr, rules(i).tailAbbr)
    Next i

    NormalizeBranchAbbr = s
End Function

'---------------------------------------------
' 事業体略語（協同組合・金融・商工会など）
'---------------------------------------------
Private Function NormalizeBusinessAbbr(ByVal s As String) As String
    Dim r As String
    r = s

    ' 協同組合系
    r = Replace(r, "農業協同組合", "ﾉｳｷﾖｳ)")
    r = Replace(r, "農協", "ﾉｳｷﾖｳ)")
    r = Replace(r, "漁業協同組合", "ｷﾞﾖｷﾖｳ)")
    r = Replace(r, "漁協", "ｷﾞﾖｷﾖｳ)")
    r = Replace(r, "生活協同組合", "ｾｲｷﾖｳ)")
    r = Replace(r, "生協", "ｾｲｷﾖｳ)")
    r = Replace(r, "協同組合", "ｷﾖｳﾄﾞｳ)")
    r = Replace(r, "事業協同組合", "ｼﾞｷﾞﾖｳ)")
    r = Replace(r, "森林組合", "ｼﾝﾘﾝ)")

    ' 金融系
    r = Replace(r, "信用金庫", "ｼﾝｷﾝ)")
    r = Replace(r, "信金", "ｼﾝｷﾝ)")
    r = Replace(r, "信用組合", "ｼﾝｸﾐ)")
    r = Replace(r, "信組", "ｼﾝｸﾐ)")
    r = Replace(r, "労働金庫", "ﾛｳｷﾝ)")
    r = Replace(r, "労金", "ﾛｳｷﾝ)")

    ' 商工会・振興会
    r = Replace(r, "商工会議所", "ｼﾖｳｺｳｶｲ)")
    r = Replace(r, "商工会", "ｼﾖｳｺｳｶｲ)")
    r = Replace(r, "振興会", "ｼﾝｺｳｶｲ)")

    NormalizeBusinessAbbr = r
End Function





'=================================================
' シート3用：口座名義／口座名義2 の自動正規化
'=================================================
Public Sub HandleAccountNormalize(ws As Worksheet, ByVal Target As Range)

    Const HEADER_ROW As Long = 3   '★見出し行は 3 行目と決め打ち

    Dim lastCol As Long
    Dim c As Long
    Dim h As String
    Dim colName1 As Long, colName2 As Long

    ' --- 1) 見出し行（3行目）の右端までを検索 ---
    lastCol = ws.Cells(HEADER_ROW, ws.Columns.Count).End(xlToLeft).Column

    For c = 1 To lastCol
        h = CStr(ws.Cells(HEADER_ROW, c).Value)

        ' 見出しのゆらぎを少し吸収
        h = Replace(h, vbCr, "")
        h = Replace(h, vbLf, "")
        h = Replace(h, "　", " ")
        h = Application.Trim(h)

        If h = "口座名義" Then
            colName1 = c
        ElseIf h = "口座名義2" Then
            colName2 = c
        End If
    Next c

    ' 見出しが1つも見つからなければ終了
    If colName1 = 0 And colName2 = 0 Then Exit Sub

    ' ヘッダー行そのものの編集なら何もしない
    If Target.Row <= HEADER_ROW Then Exit Sub

    ' 複数セル同時変更（貼り付けなど）は対象外
    If Target.CountLarge <> 1 Then Exit Sub

    ' 対象列以外は無視
    If Target.Column <> colName1 And Target.Column <> colName2 Then Exit Sub

    ' 空欄なら何もしない
    If Trim$(CStr(Target.Value)) = "" Then Exit Sub

    ' --- 2) 実際の変換 ---
    Target.Value = NormalizeAccountName(CStr(Target.Value))

End Sub
