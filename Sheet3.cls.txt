Option Explicit

Private Sub Worksheet_Change(ByVal Target As Range)
    On Error GoTo ErrHandler
    If Target.Address = Target.EntireColumn.Address Then Exit Sub
    If Target.Address = Target.EntireRow.Address Then Exit Sub

    Application.EnableEvents = False

    LogEditAndHighlight Target

    Dim headerRow As Long
    headerRow = 3
    Dim colMap As Object
    Set colMap = BuildHeaderMap(Me, headerRow)

    AutoFillVendorInfo Me, Target, colMap, headerRow
    AppendCarryOverSuffix Me, Target, colMap, headerRow
    AutoAssignNumber Me, Target, colMap, headerRow
    ReflectPeriodEnd Me, Target, colMap, headerRow
    ApplyChangePropagation Me, Target, colMap, headerRow
    ShowAmountComments Me, Target, colMap, headerRow
    NormalizeAccountInput Me, Target, colMap, headerRow

CleanExit:
    Application.EnableEvents = True
    Exit Sub

ErrHandler:
    MsgBox "エラーが発生しました：" & Err.Description, vbExclamation
    Resume CleanExit
End Sub

Private Sub AutoFillVendorInfo(ws As Worksheet, ByVal Target As Range, ByVal colMap As Object, ByVal headerRow As Long)
    Dim col業者 As Long, col起案日 As Long, col代表者 As Long, col住所 As Long
    col業者 = ColFrom(colMap, "施工業者・債権者", "業者名", "施工業者", "債権者")
    col起案日 = ColFrom(colMap, "起案日", "起案日付")
    col代表者 = ColFrom(colMap, "代表者", "代表者名", "担当者")
    col住所 = ColFrom(colMap, "住所", "所在地")

    If col業者 = 0 Or col起案日 = 0 Or col代表者 = 0 Or col住所 = 0 Then Exit Sub

    Dim rng業者 As Range
    Set rng業者 = Intersect(Target, ws.Range(ws.Cells(headerRow + 1, col業者), ws.Cells(ws.Rows.Count, col業者).End(xlUp)))
    If rng業者 Is Nothing Then Exit Sub

    Dim c As Range
    For Each c In rng業者.Cells
        If Trim(c.Value) <> "" Then
            Dim key As String, rep As String, addr As String
            key = Replace(Replace(Application.Trim(Replace(CStr(c.Value), "　", " ")), vbCr, ""), vbLf, "")
            GetLatestRepAddr ws, headerRow, col業者, col起案日, col代表者, col住所, c.Row, key, rep, addr
            If rep <> "" Then ws.Cells(c.Row, col代表者).Value = rep
            If addr <> "" Then ws.Cells(c.Row, col住所).Value = addr
        End If
    Next c
End Sub

Private Sub AppendCarryOverSuffix(ws As Worksheet, ByVal Target As Range, ByVal colMap As Object, ByVal headerRow As Long)
    Dim suffixCol As Long
    suffixCol = ColFrom(colMap, "細々節〔繰越明許分〕")
    If suffixCol = 0 Then Exit Sub

    Dim 繰越語尾 As String
    繰越語尾 = "〔繰越明許分〕"

    Dim rng As Range
    Set rng = Intersect(Target, ws.Range(ws.Cells(headerRow + 1, suffixCol), ws.Cells(1000, suffixCol)))
    If rng Is Nothing Then Exit Sub

    Dim cell As Range
    For Each cell In rng.Cells
        If Trim(cell.Value) <> "" Then
            Dim original As String, normalized As String
            original = cell.Value
            normalized = Replace(Replace(Replace(Trim(original), vbCr, ""), vbLf, ""), "　", "")
            If Right$(normalized, Len(繰越語尾)) <> 繰越語尾 Then
                cell.Value = Trim(original) & 繰越語尾
            End If
        End If
    Next cell
End Sub

Private Sub AutoAssignNumber(ws As Worksheet, ByVal Target As Range, ByVal colMap As Object, ByVal headerRow As Long)
    Dim col件名 As Long, col採番 As Long
    col件名 = ColFrom(colMap, "件名")
    col採番 = ColFrom(colMap, "採番")
    If col件名 = 0 Or col採番 = 0 Then Exit Sub

    Dim rng As Range
    Set rng = Intersect(Target, ws.Range(ws.Cells(headerRow + 1, col件名), ws.Cells(ws.Rows.Count, col件名).End(xlUp)))
    If rng Is Nothing Then Exit Sub

    Dim usedNumbers As Object
    Set usedNumbers = CreateObject("Scripting.Dictionary")

    Dim lastRow As Long, i As Long
    Dim 番号 As String
    lastRow = ws.Cells(ws.Rows.Count, col件名).End(xlUp).Row
    For i = headerRow + 1 To lastRow
        番号 = ws.Cells(i, col採番).Value
        If 番号 <> "" Then usedNumbers(CStr(Format(番号, "000"))) = True
    Next i

    Dim cell As Range, nextNumber As Long
    For Each cell In rng.Cells
        If Trim(cell.Value) <> "" And ws.Cells(cell.Row, col採番).Value = "" Then
            nextNumber = 1
            Do While usedNumbers.Exists(Format(nextNumber, "000"))
                nextNumber = nextNumber + 1
            Loop
            ws.Cells(cell.Row, col採番).Value = Format(nextNumber, "000")
            usedNumbers.Add Format(nextNumber, "000"), True
        End If
    Next cell
End Sub

Private Sub ReflectPeriodEnd(ws As Worksheet, ByVal Target As Range, ByVal colMap As Object, ByVal headerRow As Long)
    Dim pairs As Variant
    pairs = PeriodPairs()

    Dim i As Long
    For i = LBound(pairs) To UBound(pairs)
        If colMap.Exists(pairs(i)(0)) And colMap.Exists(pairs(i)(1)) And colMap.Exists(pairs(i)(2)) Then
            Dim srcCol As Long, condCol As Long, destCol As Long
            srcCol = colMap(pairs(i)(0))
            condCol = colMap(pairs(i)(1))
            destCol = colMap(pairs(i)(2))

            Dim rng As Range
            Set rng = Intersect(Target, ws.Range(ws.Cells(headerRow + 1, condCol), ws.Cells(ws.Rows.Count, condCol).End(xlUp)))
            If rng Is Nothing Then GoTo NextPair

            Dim cell As Range, rowIdx As Long, val As String
            For Each cell In rng.Cells
                rowIdx = cell.Row
                val = Trim(cell.Value)
                If val = "" Or val = "工期延期" Then
                    ws.Cells(rowIdx, destCol).Value = ""
                Else
                    ws.Cells(rowIdx, destCol).Value = ws.Cells(rowIdx, srcCol).Value
                End If
            Next cell
        End If
NextPair:
    Next i
End Sub

Private Sub ApplyChangePropagation(ws As Worksheet, ByVal Target As Range, ByVal colMap As Object, ByVal headerRow As Long)
    Dim patterns As Variant
    patterns = ChangePatterns()

    Dim col種別 As Long
    col種別 = ColFrom(colMap, "種別")

    Dim i As Long
    For i = LBound(patterns) To UBound(patterns)
        Dim condName As String, fromName As String, toName As String, changeName As String
        Dim designFrom As String, designTo As String, workFrom As String, workTo As String
        Dim amt1From As String, amt1To As String, amt2From As String, amt2To As String
        Dim amt3From As String, amt3To As String, amt4From As String, amt4To As String
        Dim suffixFrom As String

        condName = patterns(i)(0)
        fromName = patterns(i)(1)
        toName = patterns(i)(2)
        changeName = patterns(i)(3)
        designFrom = patterns(i)(4)
        designTo = patterns(i)(5)
        workFrom = patterns(i)(6)
        workTo = patterns(i)(7)
        amt1From = patterns(i)(8)
        amt1To = patterns(i)(9)
        amt2From = patterns(i)(10)
        amt2To = patterns(i)(11)
        amt3From = patterns(i)(12)
        amt3To = patterns(i)(13)
        amt4From = patterns(i)(14)
        amt4To = patterns(i)(15)
        suffixFrom = patterns(i)(16)

        If colMap.Exists(condName) Then
            Dim condCol As Long
            condCol = colMap(condName)
            Dim rng As Range
            Set rng = Intersect(Target, ws.Range(ws.Cells(headerRow + 1, condCol), ws.Cells(ws.Rows.Count, condCol).End(xlUp)))
            If rng Is Nothing Then GoTo NextPattern

            Dim cell As Range, rowIdx As Long, val As String
            For Each cell In rng.Cells
                rowIdx = cell.Row
                val = Trim(cell.Value)

                Select Case val
                    Case "増額変更", "インフレスライド"
                        If IsNumeric(ws.Cells(rowIdx, colMap(fromName)).Value) And IsNumeric(ws.Cells(rowIdx, colMap(changeName)).Value) Then
                            ws.Cells(rowIdx, colMap(toName)).Value = ws.Cells(rowIdx, colMap(fromName)).Value + ws.Cells(rowIdx, colMap(changeName)).Value
                        Else
                            ws.Cells(rowIdx, colMap(toName)).Value = ""
                        End If

                    Case "減額変更"
                        If IsNumeric(ws.Cells(rowIdx, colMap(fromName)).Value) And IsNumeric(ws.Cells(rowIdx, colMap(changeName)).Value) Then
                            ws.Cells(rowIdx, colMap(toName)).Value = ws.Cells(rowIdx, colMap(fromName)).Value - ws.Cells(rowIdx, colMap(changeName)).Value
                        Else
                            ws.Cells(rowIdx, colMap(toName)).Value = ""
                        End If

                    Case "工期延期"
                        ws.Cells(rowIdx, colMap(changeName)).Value = 0
                        ws.Cells(rowIdx, colMap(toName)).Value = ws.Cells(rowIdx, colMap(fromName)).Value
                        ws.Cells(rowIdx, colMap(designTo)).Value = ws.Cells(rowIdx, colMap(designFrom)).Value
                        ws.Cells(rowIdx, colMap(workTo)).Value = ws.Cells(rowIdx, colMap(workFrom)).Value
                        ws.Cells(rowIdx, colMap(amt1To)).Value = ws.Cells(rowIdx, colMap(amt1From)).Value
                        ws.Cells(rowIdx, colMap(amt2To)).Value = ws.Cells(rowIdx, colMap(amt2From)).Value
                        ws.Cells(rowIdx, colMap(amt3To)).Value = ws.Cells(rowIdx, colMap(amt3From)).Value
                        ws.Cells(rowIdx, colMap(amt4To)).Value = ws.Cells(rowIdx, colMap(amt4From)).Value

                    Case "軽微変更"
                        ws.Cells(rowIdx, colMap(toName)).Value = ws.Cells(rowIdx, colMap(fromName)).Value
                        ws.Cells(rowIdx, colMap(designTo)).Value = ws.Cells(rowIdx, colMap(designFrom)).Value
                        ws.Cells(rowIdx, colMap(workTo)).Value = ""
                        ws.Cells(rowIdx, colMap(amt1To)).Value = ws.Cells(rowIdx, colMap(amt1From)).Value
                        ws.Cells(rowIdx, colMap(amt2To)).Value = ws.Cells(rowIdx, colMap(amt2From)).Value
                        ws.Cells(rowIdx, colMap(amt3To)).Value = ws.Cells(rowIdx, colMap(amt3From)).Value
                        ws.Cells(rowIdx, colMap(amt4To)).Value = ws.Cells(rowIdx, colMap(amt4From)).Value
                        ws.Cells(rowIdx, colMap(changeName)).Value = ""

                    Case "丸々繰越"
                        ApplyCarryOverCopy ws, colMap, rowIdx, changeName, toName, designTo, workFrom, workTo, amt1To, amt2To, amt3To, amt4To, suffixFrom

                    Case "前金払後繰越"
                        If col種別 > 0 And Trim(ws.Cells(rowIdx, col種別).Value) = "単" Then
                            ApplyCarryOverCopy ws, colMap, rowIdx, changeName, toName, designTo, workFrom, workTo, amt1To, amt2To, amt3To, amt4To, suffixFrom
                        Else
                            ApplyCarryOverCopy ws, colMap, rowIdx, changeName, toName, designTo, workFrom, workTo, amt1To, amt2To, amt3To, amt4To, suffixFrom
                        End If

                    Case ""
                        ClearChangeValues ws, colMap, rowIdx, toName, changeName, designTo, workTo, amt1To, amt2To, amt3To, amt4To
                End Select

                If val = "増額変更" Or val = "減額変更" Or val = "工期延期" Or val = "軽微変更" Or val = "インフレスライド" Then
                    If colMap.Exists(toName) And colMap.Exists(amt1To) And colMap.Exists(amt4To) Then
                        If IsNumeric(ws.Cells(rowIdx, colMap(toName)).Value) And IsNumeric(ws.Cells(rowIdx, colMap(amt1To)).Value) Then
                            ws.Cells(rowIdx, colMap(amt4To)).Value = ws.Cells(rowIdx, colMap(toName)).Value - ws.Cells(rowIdx, colMap(amt1To)).Value
                        Else
                            ws.Cells(rowIdx, colMap(amt4To)).Value = ""
                        End If
                    End If
                End If
            Next cell
        End If
NextPattern:
    Next i
End Sub

Private Sub ApplyCarryOverCopy(ws As Worksheet, ByVal colMap As Object, ByVal rowIdx As Long, ByVal changeName As String, ByVal toName As String, ByVal designTo As String, ByVal workFrom As String, ByVal workTo As String, ByVal amt1To As String, ByVal amt2To As String, ByVal amt3To As String, ByVal amt4To As String, ByVal suffixFrom As String)
    Dim clearCols As Variant
    clearCols = Array(changeName, toName, designTo, amt1To, amt2To, amt3To, amt4To)

    Dim cname As Variant
    For Each cname In clearCols
        If colMap.Exists(cname) Then
            ws.Cells(rowIdx, colMap(cname)).Value = 0
        End If
    Next cname

    If colMap.Exists(workFrom) And colMap.Exists(workTo) Then
        ws.Cells(rowIdx, colMap(workTo)).Value = ws.Cells(rowIdx, colMap(workFrom)).Value
    End If

    Dim fld As Variant
    For Each fld In Array("始", "終", "設計金額", "債務負担額", "補助対象額", "国費率", "補助額", "単市", "工事内容")
        If colMap.Exists(fld & suffixFrom) And colMap.Exists(fld & "a") Then
            ws.Cells(rowIdx, colMap(fld & "a")).Value = ws.Cells(rowIdx, colMap(fld & suffixFrom)).Value
        End If
    Next fld
End Sub

Private Sub ClearChangeValues(ws As Worksheet, ByVal colMap As Object, ByVal rowIdx As Long, ByVal toName As String, ByVal changeName As String, ByVal designTo As String, ByVal workTo As String, ByVal amt1To As String, ByVal amt2To As String, ByVal amt3To As String, ByVal amt4To As String)
    Dim clearList As Variant
    clearList = Array(toName, changeName, designTo, workTo, amt1To, amt2To, amt3To, amt4To)

    Dim cname As Variant
    For Each cname In clearList
        If colMap.Exists(cname) Then
            ws.Cells(rowIdx, colMap(cname)).Value = ""
        End If
    Next cname
End Sub

Private Sub ShowAmountComments(ws As Worksheet, ByVal Target As Range, ByVal colMap As Object, ByVal headerRow As Long)
    If Not (colMap.Exists("最終債務負担額（合計）") And colMap.Exists("補助対象額（事業費ベース）")) Then Exit Sub

    Dim 債務列 As Long, 補助列 As Long
    債務列 = colMap("最終債務負担額（合計）")
    補助列 = colMap("補助対象額（事業費ベース）")

    If Intersect(Target, ws.Columns(債務列)) Is Nothing And Intersect(Target, ws.Columns(補助列)) Is Nothing Then Exit Sub

    Dim cell As Range
    For Each cell In Target
        If cell.Row >= headerRow + 1 Then
            Dim valW As Variant, valX As Variant, コメント内容 As String
            valW = ws.Cells(cell.Row, 債務列).Value
            valX = ws.Cells(cell.Row, 補助列).Value

            With ws.Cells(cell.Row, 1)
                If Not .Comment Is Nothing Then .Comment.Delete
                If valW <> "" And valX <> "" Then
                    コメント内容 = "最終債務：" & Format(valW, "#,##0") & "円" & vbCrLf & "補助対象：" & Format(valX, "#,##0") & "円"
                    .AddComment Text:=コメント内容
                End If
            End With
        End If
    Next cell
End Sub

Private Sub ReflectAccountName(ByVal cell As Range)
    If Trim(cell.Value) = "" Then Exit Sub
    cell.Value = NormalizeAccountName(cell.Value)
End Sub

Private Sub NormalizeAccountInput(ws As Worksheet, ByVal Target As Range, ByVal colMap As Object, ByVal headerRow As Long)
    Dim colAccount As Long
    colAccount = ColFrom(colMap, "振込先口座名義", "口座名義", "口座名義人")
    If colAccount = 0 Then Exit Sub

    Dim rng As Range
    Set rng = Intersect(Target, ws.Range(ws.Cells(headerRow + 1, colAccount), ws.Cells(ws.Rows.Count, colAccount).End(xlUp)))
    If rng Is Nothing Then Exit Sub

    Dim c As Range
    For Each c In rng.Cells
        ReflectAccountName c
    Next c
End Sub

Private Function BuildHeaderMap(ws As Worksheet, ByVal headerRow As Long) As Object
    Dim colMap As Object
    Set colMap = CreateObject("Scripting.Dictionary")

    Dim lastCol As Long, i As Long
    lastCol = ws.Cells(headerRow, ws.Columns.Count).End(xlToLeft).Column
    For i = 1 To lastCol
        colMap(ws.Cells(headerRow, i).Value) = i
    Next i

    Set BuildHeaderMap = colMap
End Function

Private Function NormalizeHeader(ByVal s As String) As String
    s = Replace(s, vbCr, "")
    s = Replace(s, vbLf, "")
    s = Replace(s, """", "")
    s = Replace(s, "　", " ")
    s = Replace(s, "（", "(")
    s = Replace(s, "）", ")")
    s = Replace(s, "〔", "[")
    s = Replace(s, "〕", "]")
    s = Application.Trim(s)
    NormalizeHeader = s
End Function

Public Function ColFrom(ByVal colMap As Object, ParamArray names() As Variant) As Long
    Dim i As Long, key As Variant
    For i = LBound(names) To UBound(names)
        If colMap.Exists(CStr(names(i))) Then
            ColFrom = colMap(CStr(names(i)))
            Exit Function
        End If
    Next i

    For Each key In colMap.Keys
        For i = LBound(names) To UBound(names)
            If NormalizeHeader(CStr(key)) = NormalizeHeader(CStr(names(i))) Then
                ColFrom = colMap(key)
                Exit Function
            End If
        Next i
    Next key
    ColFrom = 0
End Function

Public Function NormName(ByVal s As String) As String
    s = Replace(s, "　", " ")
    s = Replace(s, vbCr, "")
    s = Replace(s, vbLf, "")
    s = Application.Trim(s)
    NormName = s
End Function

Public Sub GetLatestRepAddr(ws As Worksheet, ByVal headerRow As Long, ByVal col業者 As Long, ByVal col起案日 As Long, ByVal col代表者 As Long, ByVal col住所 As Long, ByVal currentRow As Long, ByVal key As String, ByRef rep As String, ByRef addr As String)
    Dim lastRow As Long, r As Long
    Dim maxDateAny As Date, maxDateFilled As Date
    Dim repAny As String, addrAny As String

    rep = "": addr = "": maxDateAny = 0: maxDateFilled = 0
    lastRow = ws.Cells(ws.Rows.Count, col業者).End(xlUp).Row

    For r = headerRow + 1 To lastRow
        If r <> currentRow Then
            If NormName(CStr(ws.Cells(r, col業者).Value)) = NormName(key) Then
                If IsDate(ws.Cells(r, col起案日).Value) Then
                    Dim d As Date
                    d = ws.Cells(r, col起案日).Value
                    Dim repTmp As String, addrTmp As String
                    repTmp = CStr(ws.Cells(r, col代表者).Value)
                    addrTmp = CStr(ws.Cells(r, col住所).Value)

                    If d > maxDateAny Then
                        maxDateAny = d
                        repAny = repTmp
                        addrAny = addrTmp
                    End If

                    If (repTmp <> "" Or addrTmp <> "") And d > maxDateFilled Then
                        maxDateFilled = d
                        rep = repTmp
                        addr = addrTmp
                    End If
                End If
            End If
        End If
    Next r

    If rep = "" And addr = "" Then
        rep = repAny
        addr = addrAny
    End If
End Sub
