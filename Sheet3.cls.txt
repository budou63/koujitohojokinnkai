Private Sub Worksheet_Change(ByVal Target As Range)
    Dim i As Long, lastRow As Long
    Dim usedNumbers As Object
    Dim nextNumber As Long
    Dim cell As Range
    Dim 番号 As String
    Dim 件名列 As Long, 採番列 As Long, 細々節列 As Long
    Dim 繰越語尾 As String
    Dim row As Long, val As String
    Dim srcCol As Long, condCol As Long, destCol As Long
    Dim colMap As Object
    Set colMap = CreateObject("Scripting.Dictionary")

    On Error GoTo ErrorHandler
    ' ★ 行まるごとの操作（挿入・削除など）は何もしない
    If Target.Address = Target.EntireRow.Address Then Exit Sub

    Application.EnableEvents = False

    ' ▼ どの編集でもまずログ＆黄色塗り（表シートかどうかは中で判定）
    Call LogEditAndHighlight(Target)

    繰越語尾 = "〔繰越明許分〕"

    ' === 見出し名と列番号を記録 ===
    For i = 1 To Me.Cells(3, Columns.Count).End(xlToLeft).Column
        colMap(Me.Cells(3, i).value) = i
        Select Case Me.Cells(3, i).value
            Case "件名": 件名列 = i
            Case "採番": 採番列 = i
            Case "細々節〔繰越明許分〕": 細々節列 = i
        End Select
    Next i

        ' === 業者名入力時：代表者・住所を自動入力（見出し名ベースでズレない） ===
    Dim headerRow As Long: headerRow = 3
    Dim col業者 As Long, col起案日 As Long, col代表者 As Long, col住所 As Long

    ' ★あなたの見出しに合わせた候補（必要に応じて追加OK）
    col業者 = ColFrom(colMap, "施工業者・債権者", "業者名", "施工業者", "債権者")
    col起案日 = ColFrom(colMap, "起案日", "起案日付")
    col代表者 = ColFrom(colMap, "代表者", "代表者名", "担当者")   ' 担当者を代表者に流用したい場合の保険
    col住所 = ColFrom(colMap, "住所", "所在地")

    If col業者 > 0 And col起案日 > 0 And col代表者 > 0 And col住所 > 0 Then
        Dim rng業者 As Range, c As Range
        Set rng業者 = Intersect(Target, Me.Range(Me.Cells(headerRow + 1, col業者), Me.Cells(Rows.Count, col業者).End(xlUp)))
        If Not rng業者 Is Nothing Then
            For Each c In rng業者.Cells
                If Trim(c.value) <> "" Then
                    Dim key As String, rep As String, addr As String
                    key = Replace(Replace(Application.Trim(Replace(CStr(c.value), "　", " ")), vbCr, ""), vbLf, "")

                    GetLatestRepAddr Me, headerRow, col業者, col起案日, col代表者, col住所, c.row, key, rep, addr
                    If rep <> "" Then Me.Cells(c.row, col代表者).value = rep
                    If addr <> "" Then Me.Cells(c.row, col住所).value = addr
                End If
            Next c
        End If
    End If


    ' === 細々節の語尾追加 ===
    If 細々節列 > 0 Then
        If Not Intersect(Target, Me.Range(Me.Cells(4, 細々節列), Me.Cells(1000, 細々節列))) Is Nothing Then
            For Each cell In Intersect(Target, Me.Range(Me.Cells(4, 細々節列), Me.Cells(1000, 細々節列)))
                If Trim(cell.value) <> "" Then
                    Dim original As String, normalized As String
                    original = cell.value
                    normalized = Replace(Replace(Replace(Trim(original), vbCr, ""), vbLf, ""), "　", "")
                    If Right(normalized, Len(繰越語尾)) <> 繰越語尾 Then
                        cell.value = Trim(original) & 繰越語尾
                    End If
                End If
            Next cell
        End If
    End If

    ' === 採番処理 ===
    If 件名列 > 0 And 採番列 > 0 Then
        If Not Intersect(Target, Me.Range(Me.Cells(4, 件名列), Me.Cells(Rows.Count, 件名列).End(xlUp))) Is Nothing Then
            Set usedNumbers = CreateObject("Scripting.Dictionary")
            lastRow = Me.Cells(Me.Rows.Count, 件名列).End(xlUp).row
            For i = 4 To lastRow
                番号 = Me.Cells(i, 採番列).value
                If 番号 <> "" Then usedNumbers(CStr(Format(番号, "000"))) = True
            Next i
            For Each cell In Intersect(Target, Me.Range(Me.Cells(4, 件名列), Me.Cells(Rows.Count, 件名列).End(xlUp)))
                If Trim(cell.value) <> "" And Me.Cells(cell.row, 採番列).value = "" Then
                    nextNumber = 1
                    Do While usedNumbers.Exists(Format(nextNumber, "000"))
                        nextNumber = nextNumber + 1
                    Loop
                    Me.Cells(cell.row, 採番列).value = Format(nextNumber, "000")
                    usedNumbers.Add Format(nextNumber, "000"), True
                End If
            Next cell
        End If
    End If
    
    ' === 工期終了日の反映 ===
    Dim 工期ペア As Variant
    工期ペア = Array( _
        Array("終①", "変更内容②", "終②"), _
        Array("終②", "変更内容③", "終③"), _
        Array("終③", "変更内容④", "終④"), _
        Array("終a", "変更内容b", "終b"), _
        Array("終b", "変更内容c", "終c"), _
        Array("終c", "変更内容d", "終d") _
    )
    For i = LBound(工期ペア) To UBound(工期ペア)
        If colMap.Exists(工期ペア(i)(0)) And colMap.Exists(工期ペア(i)(1)) And colMap.Exists(工期ペア(i)(2)) Then
            srcCol = colMap(工期ペア(i)(0))
            condCol = colMap(工期ペア(i)(1))
            destCol = colMap(工期ペア(i)(2))
            If Not Intersect(Target, Me.Range(Me.Cells(4, condCol), Me.Cells(Rows.Count, condCol).End(xlUp))) Is Nothing Then
                For Each cell In Intersect(Target, Me.Range(Me.Cells(4, condCol), Me.Cells(Rows.Count, condCol).End(xlUp)))
                    row = cell.row
                    val = Trim(cell.value)
                    If val = "" Or val = "工期延期" Then
                        Me.Cells(row, destCol).value = ""
                    Else
                        Me.Cells(row, destCol).value = Me.Cells(row, srcCol).value
                    End If
                Next cell
            End If
        End If
    Next i

    ' === 変更内容＆丸々繰越をまとめて反映 ===
    Dim パターン As Variant
    パターン = Array( _
      Array("変更内容②", "債務負担額①", "債務負担額②", "変更額②", "設計金額①", "設計金額②", "工事内容①", "工事内容②", "補助対象額①", "補助対象額②", "国費率①", "国費率②", "補助額①", "補助額②", "単市①", "単市②", "①"), _
      Array("変更内容③", "債務負担額②", "債務負担額③", "変更額③", "設計金額②", "設計金額③", "工事内容②", "工事内容③", "補助対象額②", "補助対象額③", "国費率②", "国費率③", "補助額②", "補助額③", "単市②", "単市③", "②"), _
      Array("変更内容④", "債務負担額③", "債務負担額④", "変更額④", "設計金額③", "設計金額④", "工事内容③", "工事内容④", "補助対象額③", "補助対象額④", "国費率③", "国費率④", "補助額③", "補助額④", "単市③", "単市④", "③"), _
      Array("変更内容b", "債務負担額a", "債務負担額b", "変更額b", "設計金額a", "設計金額b", "工事内容a", "工事内容b", "補助対象額a", "補助対象額b", "国費率a", "国費率b", "補助額a", "補助額b", "単市a", "単市b", "a"), _
      Array("変更内容c", "債務負担額b", "債務負担額c", "変更額c", "設計金額b", "設計金額c", "工事内容b", "工事内容c", "補助対象額b", "補助対象額c", "国費率b", "国費率c", "補助額b", "補助額c", "単市b", "単市c", "b"), _
      Array("変更内容d", "債務負担額c", "債務負担額d", "変更額d", "設計金額c", "設計金額d", "工事内容c", "工事内容d", "補助対象額c", "補助対象額d", "国費率c", "国費率d", "補助額c", "補助額d", "単市c", "単市d", "c") _
    )

    Dim condName As String, fromName As String, toName As String, changeName As String
    Dim designFrom As String, designTo As String, workFrom As String, workTo As String
    Dim amt1From As String, amt1To As String, amt2From As String, amt2To As String
    Dim amt3From As String, amt3To As String, amt4From As String, amt4To As String
    Dim suffixFrom As String

    For i = LBound(パターン) To UBound(パターン)
        condName = パターン(i)(0)
        fromName = パターン(i)(1)
        toName = パターン(i)(2)
        changeName = パターン(i)(3)
        designFrom = パターン(i)(4)
        designTo = パターン(i)(5)
        workFrom = パターン(i)(6)
        workTo = パターン(i)(7)
        amt1From = パターン(i)(8)
        amt1To = パターン(i)(9)
        amt2From = パターン(i)(10)
        amt2To = パターン(i)(11)
        amt3From = パターン(i)(12)
        amt3To = パターン(i)(13)
        amt4From = パターン(i)(14)
        amt4To = パターン(i)(15)
        suffixFrom = パターン(i)(16)

        If colMap.Exists(condName) Then
            condCol = colMap(condName)
            If Not Intersect(Target, Me.Range(Me.Cells(4, condCol), Me.Cells(Rows.Count, condCol).End(xlUp))) Is Nothing Then
                For Each cell In Intersect(Target, Me.Range(Me.Cells(4, condCol), Me.Cells(Rows.Count, condCol).End(xlUp)))
                    row = cell.row
                    val = Trim(cell.value)

                    Select Case val
                        Case "増額変更", "インフレスライド"
                            If IsNumeric(Me.Cells(row, colMap(fromName)).value) And IsNumeric(Me.Cells(row, colMap(changeName)).value) Then
                                Me.Cells(row, colMap(toName)).value = Me.Cells(row, colMap(fromName)).value + Me.Cells(row, colMap(changeName)).value
                            Else
                                Me.Cells(row, colMap(toName)).value = ""
                            End If

                        Case "減額変更"
                            If IsNumeric(Me.Cells(row, colMap(fromName)).value) And IsNumeric(Me.Cells(row, colMap(changeName)).value) Then
                                Me.Cells(row, colMap(toName)).value = Me.Cells(row, colMap(fromName)).value - Me.Cells(row, colMap(changeName)).value
                            Else
                                Me.Cells(row, colMap(toName)).value = ""
                            End If

                        Case "工期延期"
                            Me.Cells(row, colMap(changeName)).value = 0
                            Me.Cells(row, colMap(toName)).value = Me.Cells(row, colMap(fromName)).value
                            Me.Cells(row, colMap(designTo)).value = Me.Cells(row, colMap(designFrom)).value
                            Me.Cells(row, colMap(workTo)).value = Me.Cells(row, colMap(workFrom)).value
                            Me.Cells(row, colMap(amt1To)).value = Me.Cells(row, colMap(amt1From)).value
                            Me.Cells(row, colMap(amt2To)).value = Me.Cells(row, colMap(amt2From)).value
                            Me.Cells(row, colMap(amt3To)).value = Me.Cells(row, colMap(amt3From)).value
                            Me.Cells(row, colMap(amt4To)).value = Me.Cells(row, colMap(amt4From)).value

                        Case "軽微変更"
                            Me.Cells(row, colMap(toName)).value = Me.Cells(row, colMap(fromName)).value
                            Me.Cells(row, colMap(designTo)).value = Me.Cells(row, colMap(designFrom)).value
                            Me.Cells(row, colMap(workTo)).value = ""
                            Me.Cells(row, colMap(amt1To)).value = Me.Cells(row, colMap(amt1From)).value
                            Me.Cells(row, colMap(amt2To)).value = Me.Cells(row, colMap(amt2From)).value
                            Me.Cells(row, colMap(amt3To)).value = Me.Cells(row, colMap(amt3From)).value
                            Me.Cells(row, colMap(amt4To)).value = Me.Cells(row, colMap(amt4From)).value
                            Me.Cells(row, colMap(changeName)).value = ""

                        Case "丸々繰越", "前金払後繰越"
                            Dim clearCols As Variant: clearCols = Array(changeName, toName, designTo, amt1To, amt2To, amt3To, amt4To)
                            Dim cname As Variant
                            For Each cname In clearCols
                                If colMap.Exists(cname) Then
                                    Me.Cells(row, colMap(cname)).value = 0
                                End If
                            Next cname
                            If colMap.Exists(workFrom) And colMap.Exists(workTo) Then
                                Me.Cells(row, colMap(workTo)).value = Me.Cells(row, colMap(workFrom)).value
                            End If
                            Dim fld As Variant
                            For Each fld In Array("始", "終", "設計金額", "債務負担額", "補助対象額", "国費率", "補助額", "単市", "工事内容")
                                If colMap.Exists(fld & suffixFrom) And colMap.Exists(fld & "a") Then
                                    Me.Cells(row, colMap(fld & "a")).value = Me.Cells(row, colMap(fld & suffixFrom)).value
                                End If
                            Next fld

                        Case ""
                            Dim clearList As Variant
                            clearList = Array(toName, changeName, designTo, workTo, amt1To, amt2To, amt3To, amt4To)
                            Dim cname2 As Variant
                            For Each cname2 In clearList
                                If colMap.Exists(cname2) Then
                                    Me.Cells(row, colMap(cname2)).value = ""
                                End If
                            Next cname2
                    End Select


' 単市 = 債務負担額 - 補助対象額
If val = "増額変更" Or val = "減額変更" Or val = "工期延期" Or val = "軽微変更" Or val = "インフレスライド" Then
    If colMap.Exists(toName) And colMap.Exists(amt1To) And colMap.Exists(amt4To) Then
        If IsNumeric(Me.Cells(row, colMap(toName)).value) And IsNumeric(Me.Cells(row, colMap(amt1To)).value) Then
            Me.Cells(row, colMap(amt4To)).value = Me.Cells(row, colMap(toName)).value - Me.Cells(row, colMap(amt1To)).value
        Else
            Me.Cells(row, colMap(amt4To)).value = ""
        End If
    End If
End If

                Next cell
            End If
        End If
    Next i

    ' === W列またはX列が変更されたとき：A列にコメントを表示 ===
    If colMap.Exists("最終債務負担額（合計）") And colMap.Exists("補助対象額（事業費ベース）") Then
        Dim 債務列 As Long, 補助列 As Long
        債務列 = colMap("最終債務負担額（合計）")
        補助列 = colMap("補助対象額（事業費ベース）")

        If Not Intersect(Target, Me.Columns(債務列)) Is Nothing Or Not Intersect(Target, Me.Columns(補助列)) Is Nothing Then
            For Each cell In Target
                row = cell.row
                If row >= 4 Then
                    Dim valW As Variant, valX As Variant, コメント内容 As String
                    valW = Me.Cells(row, 債務列).value
                    valX = Me.Cells(row, 補助列).value

                    With Me.Cells(row, 1) ' A列
                        ' 既存コメント削除
                        If Not .Comment Is Nothing Then .Comment.Delete

                        ' 両方に値があればコメント追加
                        If valW <> "" And valX <> "" Then
                            コメント内容 = "最終債務：" & Format(valW, "#,##0") & "円" & vbCrLf & _
                                           "補助対象：" & Format(valX, "#,##0") & "円"
                            .AddComment Text:=コメント内容
                        End If
                    End With
                End If
            Next cell
        End If
    End If

    ' === 口座名義・口座名義2列が変更されたとき：自動正規化 ===
    Call HandleAccountNormalize(Me, Target)

CleanExit:
    Application.EnableEvents = True
    Exit Sub


ErrorHandler:
    MsgBox "エラーが発生しました：" & Err.Description, vbExclamation
    Resume CleanExit
End Sub

' 選択変更時：1行目の通し番号と見出し列数を比較し、ズレていたら更新
'---------------------------------------------
Private Sub Worksheet_SelectionChange(ByVal Target As Range)
    Dim lastCol As Long
    Dim lastColInRow1 As Long
    
    On Error Resume Next
    
    ' 見出し行（3行目）の最終列を取得
    lastCol = Me.Cells(3, Me.Columns.Count).End(xlToLeft).Column
    
    ' 1行目の最終列の値を取得（空の場合は0）
    lastColInRow1 = 0
    If lastCol > 0 Then
        If IsNumeric(Me.Cells(1, lastCol).Value) Then
            lastColInRow1 = CLng(Me.Cells(1, lastCol).Value)
        End If
    End If
    
    ' 見出し列数と1行目の通し番号が一致していない場合、更新
    If lastCol <> lastColInRow1 Then
        Call RefreshColumnNumbers
    End If
    
    On Error GoTo 0
End Sub

'---------------------------------------------
' 1行目の通し番号を振り直す（見出し行の最終列まで）
'---------------------------------------------
Private Sub RefreshColumnNumbers()
    Dim lastCol As Long
    Dim c As Long
    
    ' 見出し行（3行目）の最終列を取得
    lastCol = Me.Cells(3, Me.Columns.Count).End(xlToLeft).Column
    
    ' 1行目のA列～lastCol列をクリア
    If lastCol > 0 Then
        Me.Range(Me.Cells(1, 1), Me.Cells(1, lastCol)).ClearContents
        
        ' 通し番号を設定（値のみ、関数は使わない）
        For c = 1 To lastCol
            Me.Cells(1, c).Value = c
        Next c
    End If
End Sub

Public Function NormalizeHeader(ByVal s As String) As String
    s = Replace(s, vbCr, "")
    s = Replace(s, vbLf, "")
    s = Replace(s, """", "")
    s = Replace(s, "　", " ")
    s = Replace(s, "（", "("): s = Replace(s, "）", ")")
    s = Replace(s, "〔", "["): s = Replace(s, "〕", "]")
    s = Application.Trim(s)
    NormalizeHeader = s
End Function

Public Function ColFrom(ByVal colMap As Object, ParamArray names() As Variant) As Long
    Dim i As Long, key As Variant
    ' 1) 完全一致
    For i = LBound(names) To UBound(names)
        If colMap.Exists(CStr(names(i))) Then
            ColFrom = colMap(CStr(names(i)))
            Exit Function
        End If
    Next i
    ' 2) 正規化して一致
    For Each key In colMap.Keys
        For i = LBound(names) To UBound(names)
            If NormalizeHeader(CStr(key)) = NormalizeHeader(CStr(names(i))) Then
                ColFrom = colMap(key)
                Exit Function
            End If
        Next i
    Next key
    ColFrom = 0
End Function

Public Function NormName(ByVal s As String) As String
    s = Replace(s, "　", " ")
    s = Replace(s, vbCr, "")
    s = Replace(s, vbLf, "")
    s = Application.Trim(s)
    NormName = s
End Function

Public Sub GetLatestRepAddr(ws As Worksheet, headerRow As Long, _
    col業者 As Long, col起案日 As Long, col代表者 As Long, col住所 As Long, _
    currentRow As Long, key As String, rep As String, addr As String)

    Dim lastRow As Long, r As Long
    Dim maxDateAny As Date, maxDateFilled As Date
    Dim repAny As String, addrAny As String

    rep = "": addr = "": maxDateAny = 0: maxDateFilled = 0
    lastRow = ws.Cells(ws.Rows.Count, col業者).End(xlUp).row

    For r = headerRow + 1 To lastRow
        If r <> currentRow Then
            If NormName(CStr(ws.Cells(r, col業者).value)) = NormName(key) Then
                If IsDate(ws.Cells(r, col起案日).value) Then
                    Dim d As Date: d = ws.Cells(r, col起案日).value
                    Dim repTmp As String: repTmp = CStr(ws.Cells(r, col代表者).value)
                    Dim addrTmp As String: addrTmp = CStr(ws.Cells(r, col住所).value)

                    ' 最新（中身空でも更新）
                    If d > maxDateAny Then
                        maxDateAny = d
                        repAny = repTmp
                        addrAny = addrTmp
                    End If

                    ' 「代表者or住所のどちらか入っている」最新
                    If (repTmp <> "" Or addrTmp <> "") And d > maxDateFilled Then
                        maxDateFilled = d
                        rep = repTmp
                        addr = addrTmp
                    End If
                End If
            End If
        End If
    Next r

    ' どちらも空だった場合は最新行の値（空の可能性もある）でフォールバック
    If rep = "" And addr = "" Then
        rep = repAny: addr = addrAny
    End If
End Sub
